const KEY = 'change-me-to-a-long-random-code';
function doPost(e) {
  const out = o => ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
  let b; try { b = JSON.parse(e.postData.contents); } catch (x) { return out({ error: 'bad request' }); }
  if (b.key !== KEY) return out({ error: 'not allowed · helper key ends ' + KEY.slice(-4) });
  const props = PropertiesService.getScriptProperties();
  // Jobs that change settings or send email need the admin key as well. Only project admins hold it; contractors' devices never get it.
  // Until ADMIN_KEY is set in the Script properties, the everyday key still opens them (so nothing stops working before the switch).
  const ADMIN_ACTIONS = ['sheetLink', 'summaryGet', 'summarySet', 'summaryTest', 'digestGet', 'digestSet', 'digestTest', 'mail'];
  if (ADMIN_ACTIONS.indexOf(b.action) >= 0) { const ak = props.getProperty('ADMIN_KEY'); if (ak && b.adminKey !== ak) return out({ error: 'This needs the admin key (menu > AI > Admin key). Only project admins can do it.' }); }
  if (b.action === 'ping') return out({ ok: true, version: HELPER_VERSION, adminKey: !!props.getProperty('ADMIN_KEY'), hasKey: !!(props.getProperty('CLAUDE_KEY') || props.getProperty('GEMINI_KEY')), claude: !!props.getProperty('CLAUDE_KEY'), gemini: !!props.getProperty('GEMINI_KEY') });
  // ---- digest emails built from the project's own data (see the Digest section below) ----
  if (b.action === 'digestGet') { try { return out({ ok: true, cfg: digestConfig_(b.pid) }); } catch (x) { return out({ error: 'Digest settings could not be read: ' + x.message }); } }
  if (b.action === 'digestSet') { try { return out({ ok: true, cfg: digestSet_(b) }); } catch (x) { return out({ error: 'Could not save the digest settings: ' + x.message }); } }
  if (b.action === 'digestTest') { try { return out({ ok: true, sent: digestTest_(b) }); } catch (x) { return out({ error: 'Could not send the test: ' + x.message }); } }
  // ---- an email from the app, for example an invitation (admins only, at most 5 recipients) ----
  if (b.action === 'mail') { try { const to = (b.to || []).map(x => String(x).trim()).filter(x => /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(x)).slice(0, 5); if (!to.length) return out({ error: 'No valid address' });
      MailApp.sendEmail({ to: to.join(','), subject: String(b.subject || 'Resero Site').slice(0, 200), htmlBody: String(b.html || ''), name: 'Resero Site', replyTo: String(b.replyTo || '') || undefined }); return out({ ok: true, sent: to.length }); } catch (x) { return out({ error: 'Could not send: ' + x.message }); } }
  // ---- contractor's register kept as a Google Sheet (columns Unit | Description | Contractor | Interior/Exterior? | Completed) ----
  // The sheet is fixed by the first link (Script property REGISTER_SHEET); to point at another sheet, change that property here.
  if (b.action === 'sheetLink') {
    const m = /\/d\/([a-zA-Z0-9_-]{20,})/.exec(String(b.url || '')), id = m ? m[1] : String(b.url || '').trim(), cur = props.getProperty('REGISTER_SHEET');
    if (cur && cur !== id) return out({ error: 'This helper is already linked to another sheet. Change REGISTER_SHEET in the script properties to move it.' });
    try { const ss = SpreadsheetApp.openById(id); props.setProperty('REGISTER_SHEET', id); return out({ ok: true, name: ss.getName(), rows: ss.getSheets()[0].getLastRow(), url: ss.getUrl() }); }
    catch (x) { return out({ error: 'Cannot open that sheet: ' + x.message }); }
  }
  if (b.action === 'sheetRead' || b.action === 'sheetWrite') {
    const id = props.getProperty('REGISTER_SHEET'); if (!id) return out({ error: 'nosheet' });
    let sh; try { sh = SpreadsheetApp.openById(id).getSheets()[0]; } catch (x) { return out({ error: 'Cannot open the register sheet: ' + x.message }); }
    if (b.action === 'sheetRead') { const n = sh.getLastRow(); return out({ ok: true, name: sh.getParent().getName(), rows: n ? sh.getRange(1, 1, n, 5).getDisplayValues() : [] }); }
    const norm = s => String(s || '').toLowerCase().replace(/[^a-z0-9 ]+/g, ' ').replace(/\s+/g, ' ').trim();
    const lock = LockService.getScriptLock(); lock.waitLock(20000); let set = 0, skipped = 0, added = 0;
    try {
      (b.set || []).forEach(c => { if (norm(sh.getRange(c.row, 2).getDisplayValue()) !== norm(c.desc)) { skipped++; return; } sh.getRange(c.row, 5).setValue(c.value); set++; });   // only if the row is still the same item
      const rows = (b.append || []).map(r => r.slice(0, 5));
      if (rows.length) { const start = sh.getLastRow() + 1; sh.getRange(start, 1, rows.length, 5).setValues(rows); try { sh.getRange(b.styleRow || start - 1, 1, 1, 5).copyTo(sh.getRange(start, 1, rows.length, 5), { formatOnly: true }); } catch (x) {} added = rows.length; }
    } finally { lock.releaseLock(); }
    return out({ ok: true, set, skipped, added });
  }
  // ---- daily summary email: who gets it and when is set from the app ----
  if (b.action === 'summaryGet') { try { return out({ ok: true, cfg: summaryConfig_() }); } catch (x) { return out({ error: 'Summary needs authorising: run authoriseHelper in the script editor (' + x.message + ')' }); } }
  if (b.action === 'summarySet') { try { return out({ ok: true, cfg: summarySet_(b) }); } catch (x) { return out({ error: 'Could not save the summary settings: ' + x.message }); } }
  if (b.action === 'summaryTest') { try { const to = (b.to || []).map(x => String(x).trim()).filter(x => /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(x)).slice(0, 5); return out({ ok: true, stats: summaryRun_(to, true), to: to }); } catch (x) { return out({ error: 'Could not send the test: ' + x.message }); } }
  // Claude first when a CLAUDE_KEY is set (Script property; optional CLAUDE_MODEL). If Claude fails (credit used up, outage), Gemini answers instead.
  const ck = props.getProperty('CLAUDE_KEY'), gk = props.getProperty('GEMINI_KEY'); if (!ck && !gk) return out({ error: 'nokey' });
  let claudeErr = '';
  // The admin can pick the model and effort in the app (b.model, b.effort); only known values are accepted.
  const MODELS = ['claude-opus-5-5', 'claude-sonnet-5-5', 'claude-haiku-5-5'], pickM = MODELS.indexOf(b.model) >= 0 ? b.model : null, geminiOnly = b.model === 'gemini';
  if (ck && !geminiOnly) { const c = claudeCall_(ck, pickM || props.getProperty('CLAUDE_MODEL') || 'claude-opus-5-5', b); if (c.ok) return out(c); claudeErr = c.error; if (!gk) return out({ error: 'Claude: ' + claudeErr }); }
  if (!gk) return out({ error: 'Gemini was chosen but the helper has no GEMINI_KEY.' });
  const parts = [{ text: String(b.prompt || '').slice(0, 60000) }];
  if (b.file && b.file.data) parts.push({ inline_data: { mime_type: b.file.mime || 'image/jpeg', data: b.file.data } });
  const call = model => UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/models/' + model + ':generateContent', { method: 'post', contentType: 'application/json', headers: { 'x-goog-api-key': gk }, muteHttpExceptions: true,
    payload: JSON.stringify({ contents: [{ role: 'user', parts }], generationConfig: { responseMimeType: 'application/json', temperature: 0.2 } }) });
  let model = props.getProperty('GEMINI_MODEL') || 'gemini-3.6-flash', r = call(model);
  if (r.getResponseCode() === 404) {   // model retired: pick the newest Flash model this key can use
    const list = JSON.parse(UrlFetchApp.fetch('https://generativelanguage.googleapis.com/v1beta/models?pageSize=200', { headers: { 'x-goog-api-key': gk }, muteHttpExceptions: true }).getContentText()).models || [];
    const ok = list.filter(m => (m.supportedGenerationMethods || []).indexOf('generateContent') >= 0 && /gemini-[\d.]+-flash$/.test(m.name)).map(m => m.name.replace('models/', '')).sort((a, b) => parseFloat(b.split('-')[1]) - parseFloat(a.split('-')[1]));
    if (ok.length) { model = ok[0]; props.setProperty('GEMINI_MODEL', model); r = call(model); }
  }
  const j = JSON.parse(r.getContentText() || '{}');
  if (r.getResponseCode() !== 200) return out({ error: (claudeErr ? 'Claude: ' + claudeErr + ' | ' : '') + ((j.error && j.error.message) || ('Gemini ' + r.getResponseCode())) });
  const t = (((j.candidates || [])[0] || {}).content || {}).parts || [];
  return out({ ok: true, model, via: 'gemini', fallback: claudeErr || undefined, text: t.map(p => p.text || '').join('') });
}

// One request to Claude (the Messages API). The app's prompts ask for JSON, so Claude is told to reply with JSON only,
// and anything around the JSON (a code fence, a stray sentence) is trimmed off before it goes back to the app.
function claudeCall_(key, model, b) {
  const content = [];
  if (b.file && b.file.data) content.push({ type: 'image', source: { type: 'base64', media_type: b.file.mime || 'image/jpeg', data: b.file.data } });
  content.push({ type: 'text', text: String(b.prompt || '').slice(0, 60000) });
  try {
    const r = UrlFetchApp.fetch('https://api.anthropic.com/v1/messages', { method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      headers: { 'x-api-key': key, 'anthropic-version': '2023-06-01' },
      payload: JSON.stringify(Object.assign({ model: model, max_tokens: 16000 }, ['low', 'medium', 'high', 'xhigh', 'max'].indexOf(b.effort) >= 0 ? { output_config: { effort: b.effort } } : {}, {
        system: 'You are the assistant inside a construction defects app used on New Zealand building sites. Use New Zealand English. Reply with one valid JSON value only: no code fences, no text before or after it.',
        messages: [{ role: 'user', content: content }] })) });
    const code = r.getResponseCode(), j = JSON.parse(r.getContentText() || '{}');
    if (code !== 200) return { ok: false, error: ((j.error && j.error.message) || ('HTTP ' + code)).slice(0, 200) };
    let t = (j.content || []).filter(p => p.type === 'text').map(p => p.text).join('').trim();
    t = t.replace(/^```(?:json)?\s*/i, '').replace(/\s*```$/, '');
    const a = t.search(/[\[{]/), z = Math.max(t.lastIndexOf('}'), t.lastIndexOf(']')); if (a > 0 || (z >= 0 && z < t.length - 1)) { if (a >= 0 && z > a) t = t.slice(a, z + 1); }
    return { ok: true, model: j.model || model, effort: ['low', 'medium', 'high', 'xhigh', 'max'].indexOf(b.effort) >= 0 ? b.effort : 'default', via: 'claude', text: t };
  } catch (x) { return { ok: false, error: String(x.message || x).slice(0, 200) }; }
}

function registerRows_(values) {
  const rows = [];
  for (let i = 0; i < values.length; i++) {
    const a = String(values[i][0] || '').trim(), b = String(values[i][1] || '').trim();
    if (!a || !b || /^(unit|project:?|job no\.?|access:?|date:?)$/i.test(a) || /pre-settlement inspection/i.test(b)) continue;
    const e = String(values[i][4] || '').trim();
    rows.push({ row: i + 1, unit: a, desc: b, sub: String(values[i][2] || '').trim(), ie: String(values[i][3] || '').trim(), done: /^(y|yes|completed?|done|x)$/i.test(e) });
  }
  return rows;
}

  const ss = SpreadsheetApp.openById(id), sh = ss.getSheets()[0], n = sh.getLastRow(), rows = registerRows_(n ? sh.getRange(1, 1, n, 5).getDisplayValues() : []);
  let prev = null; try { const parts = []; for (let i = 0; i < 20; i++) { const p = props.getProperty('SNAP_' + i); if (p == null) break; parts.push(p); } if (parts.length) prev = JSON.parse(parts.join('')); } catch (e) { prev = null; }
  const out = buildSummary_(rows, prev, { title: (props.getProperty('SUMMARY_TITLE') || 'Miles register') + ': daily summary', dateText: Utilities.formatDate(new Date(), SUMMARY_TZ, 'EEEE d MMMM yyyy'), span: summarySpan_(props.getProperty('SUMMARY_LAST')), url: ss.getUrl(), gid: sh.getSheetId(), name: ss.getName(), app: APP_URL, test: test });
  if (!to || !to.length) throw new Error('No one is set to receive the summary yet.');
  MailApp.sendEmail({ to: to.join(','), subject: out.subject, htmlBody: out.html, name: 'Resero Site' });
  if (!test) { props.setProperty('SUMMARY_LAST', String(Date.now())); const s = JSON.stringify(out.snapshot); for (let i = 0; i < 20; i++) props.deleteProperty('SNAP_' + i); for (let i = 0, k = 0; i < s.length; i += 8000, k++) props.setProperty('SNAP_' + k, s.slice(i, i + 8000)); }
  return out.stats;
}

// ================= Digest emails from the project's own data =================
// Each morning (7am New Zealand time) the helper reads the project from Firestore with the script owner's Google access and emails:
//  - inspectors (daily): items in their registers newly marked ready for review, each linked straight to the item;
//  - contractors (daily): items sent back, items added, items past their due date, open items by subcontractor;
//  - admins (weekly, Mondays): the week's movement, what is overdue, warranty submissions waiting, maintenance coming up;
//  - the client (monthly, first weekday): what is overdue, due this month and in the next three months.
// Who gets what is set in the app (menu > Email digests) and kept in Script properties (DIGEST_<project id>).
const FS_PROJECT = 'resero-site-tools';
function fsVal_(v) { if (!v) return null; if ('stringValue' in v) return v.stringValue; if ('integerValue' in v) return Number(v.integerValue); if ('doubleValue' in v) return v.doubleValue; if ('booleanValue' in v) return v.booleanValue; if ('nullValue' in v) return null; if ('timestampValue' in v) return v.timestampValue; if ('arrayValue' in v) return (v.arrayValue.values || []).map(fsVal_); if ('mapValue' in v) { const o = {}; Object.keys(v.mapValue.fields || {}).forEach(k => { o[k] = fsVal_(v.mapValue.fields[k]); }); return o; } return null; }
function fsFetch_(url) { const r = UrlFetchApp.fetch(url, { headers: { Authorization: 'Bearer ' + ScriptApp.getOAuthToken() }, muteHttpExceptions: true }); if (r.getResponseCode() !== 200) throw new Error('Reading the project failed (' + r.getResponseCode() + '): ' + r.getContentText().slice(0, 160)); return JSON.parse(r.getContentText()); }
const FS_BASE = () => 'https://firestore.googleapis.com/v1/projects/' + FS_PROJECT + '/databases/(default)/documents/';
function fsGet_(path) { const d = fsFetch_(FS_BASE() + path); return Object.assign(fsVal_({ mapValue: { fields: d.fields || {} } }), { id: d.name.split('/').pop() }); }
function fsList_(path) { const out = []; let tok = ''; do { const j = fsFetch_(FS_BASE() + path + '?pageSize=300' + (tok ? '&pageToken=' + encodeURIComponent(tok) : '')); (j.documents || []).forEach(d => out.push(Object.assign(fsVal_({ mapValue: { fields: d.fields || {} } }), { id: d.name.split('/').pop() }))); tok = j.nextPageToken || ''; } while (tok); return out; }
function digestAll_() { const p = PropertiesService.getScriptProperties().getProperties(); return Object.keys(p).filter(k => /^DIGEST_/.test(k)).map(k => { try { return JSON.parse(p[k]); } catch (e) { return null; } }).filter(Boolean); }
function digestConfig_(pid) { const v = PropertiesService.getScriptProperties().getProperty('DIGEST_' + pid); return v ? JSON.parse(v) : { pid: pid, recipients: [] }; }
function digestSet_(b) { if (!b.pid) throw new Error('No project'); const rec = (b.recipients || []).filter(r => r && /^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(r.email || '')).slice(0, 40).map(r => ({ email: String(r.email).toLowerCase(), name: String(r.name || ''), kind: ['inspector', 'contractor', 'admin', 'client'].indexOf(r.kind) >= 0 ? r.kind : 'contractor', freq: ['daily', 'weekly', 'monthly', 'off'].indexOf(r.freq) >= 0 ? r.freq : 'daily', registers: (r.registers || []).map(String).slice(0, 20) }));
  const cfg = { pid: String(b.pid), name: String(b.name || ''), app: String(b.app || APP_URL), recipients: rec }; PropertiesService.getScriptProperties().setProperty('DIGEST_' + cfg.pid, JSON.stringify(cfg)); ensureDigestTrigger_(); return cfg; }
function ensureDigestTrigger_() { if (!ScriptApp.getProjectTriggers().some(t => t.getHandlerFunction() === 'digestDaily')) ScriptApp.newTrigger('digestDaily').timeBased().everyDays(1).atHour(7).inTimezone(SUMMARY_TZ).create(); }
function digestTest_(b) { const cfg = digestConfig_(b.pid), r = (cfg.recipients || []).find(x => x.email === String(b.email || '').toLowerCase()); if (!r) throw new Error('That person is not on the digest list yet. Save the list first.');
  const to = String(b.to || '').trim(); if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(to)) throw new Error('No address to send the test to'); const data = digestData_(cfg.pid), m = digestBuild_(cfg, r, data, true); MailApp.sendEmail({ to: to, subject: 'TEST: ' + m.subject, htmlBody: m.html, name: 'Resero Site' }); return to; }
function digestDaily() { const now = new Date(), dow = parseInt(Utilities.formatDate(now, SUMMARY_TZ, 'u'), 10), dom = parseInt(Utilities.formatDate(now, SUMMARY_TZ, 'd'), 10); if (dow > 5) return;
  const firstWeekday = dom <= 3 && (dom === 1 || dow === 1);
  digestAll_().forEach(cfg => { let data = null; (cfg.recipients || []).forEach(r => { const due = r.freq === 'daily' || (r.freq === 'weekly' && dow === 1) || (r.freq === 'monthly' && firstWeekday); if (!due) return;
    try { data = data || digestData_(cfg.pid); const m = digestBuild_(cfg, r, data, false); if (m.skip) return; MailApp.sendEmail({ to: r.email, subject: m.subject, htmlBody: m.html, name: 'Resero Site' }); PropertiesService.getScriptProperties().setProperty('DLAST_' + cfg.pid + '_' + r.email, String(Date.now())); } catch (e) { console.warn('digest ' + r.email + ': ' + e.message); } }); }); }
function digestData_(pid) { const p = fsGet_('projects/' + pid); return { p: p, D: fsList_('projects/' + pid + '/defects'), docs: (() => { try { return fsList_('projects/' + pid + '/docs'); } catch (e) { return []; } })(), T: (() => { try { return fsList_('projects/' + pid + '/tasks'); } catch (e) { return []; } })(), A: (() => { try { return fsList_('projects/' + pid + '/assets'); } catch (e) { return []; } })() }; }
const dIso_ = d => Utilities.formatDate(d, SUMMARY_TZ, 'yyyy-MM-dd');
const dNz_ = iso => { if (!iso) return ''; const d = new Date(String(iso).slice(0, 10) + 'T00:00:00'); return Utilities.formatDate(d, SUMMARY_TZ, 'd MMM yyyy'); };
const addM_ = (iso, n) => { const d = new Date(String(iso).slice(0, 10) + 'T00:00:00'); d.setMonth(d.getMonth() + n); return dIso_(d); };
const esc_ = s => String(s == null ? '' : s).replace(/[&<>"]/g, c => ({ '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;' }[c]));
function digestBuild_(cfg, r, data, test) {
  const props = PropertiesService.getScriptProperties(), last = parseInt(props.getProperty('DLAST_' + cfg.pid + '_' + r.email) || '0', 10);
  const span = r.freq === 'weekly' ? 7 : r.freq === 'monthly' ? 31 : 1, since = new Date(last && !test ? last : Date.now() - span * 864e5).toISOString(), today = dIso_(new Date());
  const app = cfg.app || APP_URL, link = (id, t) => '<a href="' + app + '#id=' + encodeURIComponent(id) + '" style="color:#1f5fa8">' + esc_(t || id) + '</a>';
  const D = data.D, inReg = d => !(r.registers || []).length || (r.registers || []).indexOf(d.source || '') >= 0, open = d => d.status !== 'Closed';
  const moved = (d, rx) => (d.history || []).some(h => String(h.t || '') > since && (rx.test(String(h.txt || '')) || (h.st && rx.test(String(h.st[1] || '')))));
  const H = (t, n) => '<h3 style="font:600 15px Arial;margin:18px 0 6px;color:#011c35">' + esc_(t) + (n != null ? ' <span style="color:#556">(' + n + ')</span>' : '') + '</h3>';
  const ul = (L, max) => '<ul style="margin:0;padding-left:18px;font:14px Arial">' + L.slice(0, max || 25).map(x => '<li style="margin:2px 0">' + x + '</li>').join('') + (L.length > (max || 25) ? '<li>and ' + (L.length - (max || 25)) + ' more in the app</li>' : '') + '</ul>';
  const box = (n, l, c) => '<td style="padding:10px 14px;border:1px solid #dde3ea;border-radius:8px;text-align:center"><div style="font:700 22px Arial;color:' + (c || '#011c35') + '">' + n + '</div><div style="font:12px Arial;color:#556">' + esc_(l) + '</div></td>';
  const tiles = L => '<table cellspacing="6" style="margin:8px 0"><tr>' + L.map(x => box(x[0], x[1], x[2])).join('') + '</tr></table>';
  const item = d => link(d.id, d.id) + ' ' + esc_((d.title || d.desc || '').slice(0, 80)) + (d.unit ? ' · Unit ' + esc_(d.unit) : '');
  let title = '', body = '', skip = false;
  if (r.kind === 'inspector') {
    const mine = D.filter(inReg), wait = mine.filter(d => d.status === 'Ready for review'), fresh = wait.filter(d => moved(d, /^Ready for (inspection|review)/i));
    title = fresh.length + ' new to inspect, ' + wait.length + ' waiting'; skip = !test && !wait.length;
    body = tiles([[fresh.length, 'newly ready for review', '#2f78c4'], [wait.length, 'waiting for you in total'], [mine.filter(d => d.status === 'Open').length, 'open with the contractor']])
      + (fresh.length ? H('Newly ready for review', fresh.length) + ul(fresh.map(item)) : '<p style="font:14px Arial">Nothing new since the last email.</p>')
      + '<p style="font:14px Arial;margin-top:14px"><a href="' + app + '#inspect=1" style="color:#1f5fa8;font-weight:600">Open everything waiting for you</a></p>';
  } else if (r.kind === 'contractor') {
    const mine = D.filter(inReg), back = mine.filter(d => open(d) && moved(d, /^Reopened|sent back/i)), added = mine.filter(d => String(d.created || '') > since), over = mine.filter(d => open(d) && d.status !== 'Ready for review' && d.targetDate && d.targetDate < today);
    const subs = {}; mine.filter(d => d.status === 'Open' || d.status === 'In progress').forEach(d => { const k = d.subcontractor || 'No subcontractor named'; subs[k] = (subs[k] || 0) + 1; });
    title = back.length + ' sent back, ' + added.length + ' new, ' + over.length + ' overdue';
    body = tiles([[mine.filter(d => d.status === 'Open').length, 'open', '#d64545'], [mine.filter(d => d.status === 'In progress').length, 'in progress', '#e39b1b'], [mine.filter(d => d.status === 'Ready for review').length, 'waiting for inspection', '#2f78c4'], [over.length, 'past due date', over.length ? '#d64545' : '#2e9b5f']])
      + (back.length ? H('Sent back to you', back.length) + ul(back.map(item)) : '') + (added.length ? H('New items', added.length) + ul(added.map(item)) : '') + (over.length ? H('Past their due date', over.length) + ul(over.sort((a, b) => String(a.targetDate).localeCompare(String(b.targetDate))).map(d => item(d) + ' · due ' + dNz_(d.targetDate)), 15) : '')
      + H('Open items by subcontractor') + ul(Object.keys(subs).sort((a, b) => subs[b] - subs[a]).map(k => esc_(k) + ': ' + subs[k]), 12);
  } else if (r.kind === 'admin') {
    const wk = d => (d.history || []).filter(h => String(h.t || '') > since), raised = D.filter(d => String(d.created || '') > since), closed = D.filter(d => d.status === 'Closed' && moved(d, /Closed|Accepted and closed|Completed and closed/i)), rfr = D.filter(d => moved(d, /^Ready for (inspection|review)/i)), reo = D.filter(d => moved(d, /^Reopened/i));
    const over = D.filter(d => open(d) && d.targetDate && d.targetDate < today), byReg = {}; D.filter(open).forEach(d => { byReg[d.source || 'Other'] = (byReg[d.source || 'Other'] || 0) + 1; });
    const subs = data.docs.filter(x => x.sub && x.sub.status === 'Awaiting review').length, tDue = data.T.filter(t => t.nextDue && t.nextDue >= today && t.nextDue <= addM_(today, 1)).length, tOver = data.T.filter(t => t.nextDue && t.nextDue < today).length;
    title = 'Week: ' + raised.length + ' raised, ' + closed.length + ' closed, ' + D.filter(open).length + ' still open';

      + H('Open by register') + ul(Object.keys(byReg).sort().map(k => esc_(k) + ': ' + byReg[k])) + (over.length ? H('Past their due date', over.length) + ul(over.map(item), 15) : '')
      + H('Handover and maintenance') + ul([subs + ' warranty submission' + (subs === 1 ? '' : 's') + ' waiting for your review', tOver + ' maintenance task' + (tOver === 1 ? '' : 's') + ' overdue, ' + tDue + ' due in the next month']);

    if (p.bwofDue) items.push({ d: p.bwofDue, t: 'Building warrant of fitness due, with the IQP Form 12A certificates' });
    if (pc) { const end = addM_(pc, +(p.dnpMonths || 12)); items.push({ d: end, t: 'Defects notification period ends: defects must be notified before this date' }); items.push({ d: addM_(end, -1), t: 'Final defects walk-through, before the period ends' }); }
